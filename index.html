💾 PostGIS Core Database Schema (DDL)

-- PostGIS Schema for Geo-Agricultural Problem & Advisory Engine CREATE EXTENSION IF NOT EXISTS postgis; CREATE EXTENSION IF NOT EXISTS "uuid-ossp"; CREATE TABLE geo_regions ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), country_code VARCHAR(3) NOT NULL, state_province VARCHAR(100) NOT NULL, agro_climatic_classification VARCHAR(100), boundary GEOMETRY(MultiPolygon, 4326) ); CREATE TABLE regional_problems ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), geo_region_id UUID REFERENCES geo_regions(id) ON DELETE CASCADE, threat_category VARCHAR(50) NOT NULL, threat_title VARCHAR(255) NOT NULL, severity_level VARCHAR(20) NOT NULL, root_cause_scientific TEXT NOT NULL, immediate_farmer_action TEXT NOT NULL, technological_intervention TEXT NOT NULL, policy_subsidy_framework TEXT NOT NULL ); CREATE INDEX idx_geo_boundary ON geo_regions USING GIST(boundary);
